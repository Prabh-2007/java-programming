import java.util.*;
public class Main
{
    public static int value(int size,int arr[]){
        int evensum=0;
        int oddsum=0;
        for(int i=0;i<size;i++){
            if(arr[i]%2==0){
                evensum+=arr[i];
            }
            else{
                oddsum+=arr[i];
            }
        }
        int result=evensum-oddsum;
        if(result>=0){
                return result;
            }
            else{
                return result*-1;
            }
    }
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int size=sc.nextInt();
		int arr[]=new int[size];
		for(int i=0;i<size;i++){
		    arr[i]=sc.nextInt();
		}
		int ans=value(size,arr);
		System.out.println(ans);
	}
}
