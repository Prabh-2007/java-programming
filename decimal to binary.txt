import java.util.Scanner;
class DecimalToBinary{
    static int DecimaToBinary(int decimal){
        int binary = 0;
        int place = 1;
        while(decimal > 0){
            int r = decimal % 2;
            binary = binary + r * place;
            place = place * 10;
            decimal = decimal / 2;
        }
        return binary;
    }
    public static void main(String args[]){
        Scanner sc = new Scanner(System.in);
        int decimal = sc.nextInt();
        System.out.print(DecimaToBinary(decimal));
    }
}

output:

i/p 2
o/p 10
