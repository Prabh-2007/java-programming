public class Student {
    String name;
    int age;

    Student() {
        name = "Prabhavathi";
        age = 19;
    }

    void display() {
        System.out.println("Name: " + name);
        System.out.println("Age: " + age);
    }

    public static void main(String[] args) {
        Student obj = new Student();
        obj.display();
    }
}
