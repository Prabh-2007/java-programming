import java.util.Scanner;
class Main {
    boolean isPrime(int n) {
        if (n <= 1)
            return false;
        for (int i = 2; i <= n / 2; i++) {
            if (n % i == 0)
                return false;
        }
        return true;
    }
    int reverse(int n) {
        int rev = 0;
        while (n > 0) {
            int d = n % 10;
            rev = rev * 10 + d;
            n = n / 10;
        }
        return rev;
    }
    void twistedPrime(int n) {
        int rev = reverse(n);
        if (isPrime(n) && isPrime(rev))
            System.out.print("Twisted Prime");
        else
            System.out.print("Not Twisted Prime");
    }
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Main obj = new Main();
        obj.twistedPrime(n);
    }
}
