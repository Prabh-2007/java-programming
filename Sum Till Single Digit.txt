import java.util.Scanner;

class Main {

    void singleDigitSum(int n) {

        while (n >= 10) {

            int sum = 0;

            while (n > 0) {

                int rem = n % 10;
                sum = sum + rem;
                n = n / 10;
            }

            n = sum;
        }

        System.out.println(n);
    }

    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        Main obj = new Main();
        obj.singleDigitSum(n);
    }
}
