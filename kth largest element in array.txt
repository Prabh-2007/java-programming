import java.util.*;
public class Main{
    public static void value(int size,int arr[],int key){
        for(int i=0;i<size-1;i++){
            for(int j=0;j<size-i-1;j++){
                if(arr[j]<arr[j+1]){
                int temp=arr[j];
                arr[j]=arr[j+1];
                arr[j+1]=temp;
            }}
        }
        System.out.println(arr[key-1]);
    }
    public static void main (String[] args) {
        /* code */
        Scanner sc=new Scanner(System.in);
        int size=sc.nextInt();
        int arr[]=new int[size];
        for(int i=0;i<size;i++){
            arr[i]=sc.nextInt();
        }
        int key=sc.nextInt();
        value(size,arr,key);
    }
}
