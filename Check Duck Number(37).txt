import java.util.Scanner;

public class DuckNumber {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int num = sc.nextInt();
        int temp = num;
        boolean duck = false;

        while (temp != 0) {
            int digit = temp % 10;

            if (digit == 0) {
                duck = true;
                break;
            }

            temp /= 10;
        }

        if (duck)
            System.out.println("Duck Number");
        else
            System.out.println("Not a Duck Number");
    }
}
